32:       throw new Error('state directory has unsafe owner or type')
33:     }

35:     // Refuse — deliberately do NOT chmod and carry on. If another account could write here, it could
36:     // already have planted the registry, the token or the hook credential, and tightening afterwards
37:     // would leave us trusting whatever it left behind. Three tests pin this (registry.spec.ts,
38:     // hookAuth.spec.ts, terminalConfigSnapshot.spec.ts); the fix belongs at creation time, and every
39:     // site that makes a directory under ADAPTER_DATA_DIR now passes mode 0o700 so Ubuntu's default
40:     // umask 0002 can no longer produce a 0775 one. This branch is what a PRE-EXISTING 0775 directory
41:     // (made by an older build on Ubuntu) lands in, so the message has to say what to do about it.
42:     if ((mode & 0o022) !== 0) {
43:       throw new Error(`state directory ${directory} is group/world writable, so its contents cannot be`
44:         + ` trusted. If you did not set that mode yourself, delete it; otherwise: chmod 700 ${directory}`)
45:     }
